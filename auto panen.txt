do
    local dragging, dragStart, startPos
    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = main.Position

                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                      startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function setStatus(t) statusLabel.Text = "Status: " .. t end

local function getChar()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if hum and root and hum.Health > 0 then return char, hum, root end
end

local function alive(id) return running and runId == id end

local function currentElapsed()
    local e = stats.elapsed
    if running and stats.runStart then e += os.clock() - stats.runStart end
    return e
end

local function updateStats()
    local e = math.floor(currentElapsed())
    local timeStr
    if e >= 3600 then
        timeStr = string.format("%d:%02d:%02d", e // 3600, (e % 3600) // 60, e % 60)
    else
        timeStr = string.format("%02d:%02d", e // 60, e % 60)
    end
    statsLabel.Text = string.format("🌱 Panen: %d   |   🔁 Loop: %d   |   ⏱ %s",
        stats.harvested, stats.loops, timeStr)
end

local function updateChestInfo()
    local full = 0
    for _, c in ipairs(chests) do if c.full then full += 1 end end
    chestInfo.Text = string.format("Peti: %d | Penuh: %d | Tersimpan: %d item",
        #chests, full, stats.deposited)
end

    updateChestInfo()

local function parseKeywords(str)
    local out = {}
    for part in string.gmatch(str or "", "[^,]+") do
        local k = part:match("^%s*(.-)%s*$"):lower()
        if k ~= "" then table.insert(out, k) end
    end
    return out
end

local function nameMatches(name, kws)
    local n = name:lower()
    for _, k in ipairs(kws) do
        if n:find(k, 1, true) then return true end
    end
    return false
end

-- khusus peti: bisa cocok persis (exactMatch = 1) atau mengandung kata kunci
local function chestNameMatches(name, kws)
    if config.exactMatch == 1 then
        local n = name:lower()
        for _, k in ipairs(kws) do
            if n == k then return true end
        end
        return false
    end
    return nameMatches(name, kws)
end

local function isInsideCharacter(inst)
    for _, pl in ipairs(Players:GetPlayers()) do
        local ch = pl.Character
        if ch and (inst == ch or inst:IsDescendantOf(ch)) then return true end
    end
    return false
end

local function getPart(inst)
    if inst:IsA("BasePart") then return inst end
    if inst:IsA("Model") then
        return inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true)
    end
end

local function chestPos(chest)
    local ok, cf = pcall(function() return chest.inst:GetPivot() end)
    if ok and cf then return cf.Position end
    return chest.part.Position
end

local function chestValid(chest)
    return chest.inst and chest.inst.Parent ~= nil and chest.part and chest.part.Parent ~= nil
end

local function removeMarker(chest)
    if chest.marker then chest.marker:Destroy() chest.marker = nil end
end

local function updateMarker(chest)
    if not chest.marker or not chest.label then return end
    if chest.full then
        chest.label.BackgroundColor3 = COLORS.red
        chest.label.Text = chest.number .. " ✕"
    else
        chest.label.BackgroundColor3 = COLORS.chest
        chest.label.Text = tostring(chest.number)
    end
end

local function scanChests()
    local _, _, root = getChar()
    if not root then setStatus("Karakter tidak ditemukan"); return 0 end

    local kws = parseKeywords(config.chestKeywords)
    if #kws == 0 then setStatus("Isi nama peti dulu"); return 0 end

    for i = #chests, 1, -1 do
        if not chestValid(chests[i]) then
            removeMarker(chests[i])
            table.remove(chests, i)
        end
    end
    local known = {}
    for _, c in ipairs(chests) do known[c.inst] = true end

    local matched, found = {}, {}
    for _, inst in ipairs(workspace:GetDescendants()) do
        if (inst:IsA("Model") or inst:IsA("BasePart"))
            and chestNameMatches(inst.Name, kws) and not isInsideCharacter(inst) then
            local skip = false
            local p = inst.Parent
            while p and p ~= workspace do
                if matched[p] then skip = true break end
                p = p.Parent
            end
            if not skip then
                matched[inst] = true
                if not known[inst] then
                    local part = getPart(inst)
                    if part then
                        local d = (inst:GetPivot().Position - root.Position).Magnitude
                        if d <= config.chestScanRadius then
                            table.insert(found, { inst = inst, part = part, dist = d })
                        end
                    end
                end
            end
        end
    end

    table.sort(found, function(a, b) return a.dist < b.dist end)

    for _, f in ipairs(found) do
        local chest = { inst = f.inst, part = f.part, number = nextChestNo, full = false }
        nextChestNo += 1
        makeMarker(chest)
        table.insert(chests, chest)
    end

    refreshChestList()
    updateChestInfo()
    return #found
end

local function resetChests()
    for _, c in ipairs(chests) do removeMarker(c) end
    chests = {}
    nextChestNo = 1
    depositCooldown = 0
    refreshChestList()
    updateChestInfo()
end

        setStatus("Tidak ada Trunk ditemukan, cek nama/radius")
    else
        setStatus("Tidak ada peti baru")
    end
end)

local AMOUNT_NAMES = { "Amount", "Count", "Quantity", "Stack" }

local function countTool(tool)
    for _, n in ipairs(AMOUNT_NAMES) do
        local v = tool:GetAttribute(n)
        if type(v) == "number" then return v end
    end
    for _, n in ipairs(AMOUNT_NAMES) do
        local v = tool:FindFirstChild(n)
        if v and (v:IsA("IntValue") or v:IsA("NumberValue")) then return v.Value end
    end
    return 1
end

local function allTools()
    local list = {}
    local bp = player:FindFirstChildOfClass("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") then table.insert(list, t) end
        end
    end
    local char = player.Character
    if char then
        for _, t in ipairs(char:GetChildren()) do
            if t:IsA("Tool") then table.insert(list, t) end
        end
    end
    return list
end

local function bagIsFull()
    if CUSTOM_BAG_FULL then
        local ok, res = pcall(CUSTOM_BAG_FULL)
        return ok and res == true
    end
    local total = 0
    for _, t in ipairs(allTools()) do total += countTool(t) end
    return total >= config.bagCapacity
end

local function collectMaterials(skipTool)
    local matKw  = parseKeywords(config.materialKeywords)
    local keepKw = parseKeywords(config.keepKeywords)
    local out = {}
    for _, t in ipairs(allTools()) do
        local ok = true
        if #matKw > 0 then
            ok = nameMatches(t.Name, matKw)
        elseif skipTool and t == skipTool then
            ok = false -- alat panen yang dipegang tidak ikut disimpan
        end
        if ok and #keepKw > 0 and nameMatches(t.Name, keepKw) then ok = false end
        if ok then table.insert(out, t) end
    end
    return out
end

local function countMaterials(skipTool)
    if CUSTOM_MATERIAL_COUNT then
        local ok, n = pcall(CUSTOM_MATERIAL_COUNT)
        return ok and tonumber(n) or 0
    end
    local total = 0
    for _, t in ipairs(collectMaterials(skipTool)) do total += countTool(t) end
    return total
end

local function pressKey(keyCode, hold)
    pcall(function()
        VIM:SendKeyEvent(true, keyCode, false, game)
        task.wait(hold or 0.05)
        VIM:SendKeyEvent(false, keyCode, false, game)
    end)
end

local function pressE() pressKey(Enum.KeyCode.E, 0.05) end

local function openKeyCode()
    local k = tostring(config.openKey or "F"):upper()
    local ok, kc = pcall(function() return Enum.KeyCode[k] end)
    return (ok and kc) or Enum.KeyCode.F
end

-- seret dari posisi layar A ke B
local function dragMouse(from, to)
    pcall(function()
        VIM:SendMouseMoveEvent(from.X, from.Y, game)
        task.wait(0.08)
        VIM:SendMouseButtonEvent(from.X, from.Y, 0, true, game, 0)
        task.wait(0.12)
        local steps = 10
        for i = 1, steps do
            local p = from:Lerp(to, i / steps)
            VIM:SendMouseMoveEvent(p.X, p.Y, game)
            task.wait(0.02)
        end
        task.wait(0.08)
        VIM:SendMouseButtonEvent(to.X, to.Y, 0, false, game, 0)
    end)
end

local function walkTo(target, myId, radius)
    radius = radius or config.radius
    local start = os.clock()
    local lastPos, lastCheck = nil, os.clock()

    while alive(myId) do
        local _, hum, root = getChar()
        if not root then return false end

        local flat = Vector3.new(root.Position.X - target.X, 0, root.Position.Z - target.Z).Magnitude
        if flat <= radius and math.abs(root.Position.Y - target.Y) < 8 then
            return true
        end
        if os.clock() - start > config.stepTimeout then
            return false
        end

        hum:MoveTo(target)

        if os.clock() - lastCheck >= 1 then
            if lastPos and (root.Position - lastPos).Magnitude < 0.6 then
                hum.Jump = true
            end
            lastPos = root.Position
            lastCheck = os.clock()
        end
        task.wait(0.1)
    end
    return false
end

local function waitInterruptible(seconds, myId)
    local t = os.clock()
    while alive(myId) and os.clock() - t < seconds do
        task.wait(0.1)
    end
end

local function pickChest()
    for _, c in ipairs(chests) do
        if not c.full and chestValid(c) then return c end
    end
end

local function slotPos(idx)
    if calib.step and idx > 1 then
        return calib.slot + calib.step * (idx - 1)
    end
    return calib.slot
end

-- Mengembalikan: "ok", "allfull", "nochest", "nocalib", atau "stopped"
local function depositAll(myId)
    local char, hum, root = getChar()
    if not root then return "stopped" end

    if not (calib.slot and calib.drop) then
        setStatus("Atur Kalibrasi Drag dulu (tab Peti)")
        depositCooldown = os.clock() + 30
        return "nocalib"
    end

    if #chests == 0 then
        setStatus("Mencari Trunk terdekat...")
        scanChests()
        if #chests == 0 then return "nochest" end
    end

    local returnPos = root.Position
    local skipTool = char:FindFirstChildOfClass("Tool")
    local result = "ok"
    local deposited = 0

    while alive(myId) do
        if countMaterials(skipTool) <= 0 then break end

        local chest = pickChest()
        if not chest then result = "allfull" break end

        -- menuju peti & buka dengan F
        setStatus(string.format("Menuju peti #%d", chest.number))
        local standDist = math.max(2, config.chestDist)
        walkTo(chestPos(chest), myId, standDist)
        if not alive(myId) then break end

        setStatus(string.format("Membuka peti #%d (%s)", chest.number, config.openKey))
        pressKey(openKeyCode(), 0.1)
        waitInterruptible(config.openDelay, myId)

        local idx, fails = 1, 0
        while alive(myId) and not chest.full do
            local before = countMaterials(skipTool)
            if before <= 0 then break end

            setStatus(string.format("Pindah item ke peti #%d (slot %d)", chest.number, idx))
            dragMouse(slotPos(idx), calib.drop)
            waitInterruptible(config.depositDelay, myId)
            local after = countMaterials(skipTool)

            if after < before then
                local n = before - after
                deposited += n
                stats.deposited += n
                fails = 0
                updateChestInfo()
            else
                fails += 1
                if fails >= 2 then
                    if calib.step and idx < config.slotCount then
                        idx += 1
                        fails = 0
                    else
                        chest.full = true
                        updateMarker(chest)
                        updateChestInfo()
                        refreshChestList()
                        setStatus(string.format("Peti #%d penuh, pindah peti berikutnya", chest.number))
                    end
                end
            end
        end

        if config.closeChest == 1 and alive(myId) then
            pressKey(openKeyCode(), 0.1)
            task.wait(0.3)
        end
    end

    if alive(myId) then
        setStatus("Kembali ke rute...")
        walkTo(returnPos, myId, config.radius + 2)
    end

    if not alive(myId) then return "stopped" end
    if deposited == 0 and result ~= "allfull" then
        depositCooldown = os.clock() + 20
    end
    return result
end

local function stopRun(msg)
    if running and stats.runStart then
        stats.elapsed += os.clock() - stats.runStart
    end
    stats.runStart = nil
    running = false
    runId += 1
    local _, hum, root = getChar()
    if hum and root then hum:MoveTo(root.Position) end
    highlight(nil)
    updateButtons()
    setStatus(msg or "Berhenti")
end

local function startRun()
    if running then return end
    if calibStep > 0 then setStatus("Selesaikan kalibrasi dulu"); return end
    if #points == 0 then setStatus("Tambahkan titik dulu!"); return end

    running = true
    runId += 1
    local myId = runId
    stats.runStart = os.clock()
    updateButtons()

    task.spawn(function()
        local loopCount = 0
        while alive(myId) do
            for i, pt in ipairs(points) do
                if not alive(myId) then break end

                if config.autoStore and os.clock() >= depositCooldown and bagIsFull() then
                    setStatus("Tas penuh, menyimpan ke Trunk...")
                    local res = depositAll(myId)
                    if res == "allfull" then
                        stopRun("Semua peti penuh, script berhenti")
                        return
                    elseif res == "nochest" then
                        depositCooldown = os.clock() + 30
                        setStatus("Tas penuh tapi tidak ada Trunk terdeteksi")
                    end
                    if not alive(myId) then break end
                end

                highlight(i)
                setStatus(string.format("Loop %d | Menuju titik %d/%d", loopCount + 1, i, #points))

                walkTo(pt.pos, myId)

                if alive(myId) and pt.type == "action" then
                    task.wait(0.2)
                    pressE()
                    stats.harvested += 1
                    updateStats()
                    setStatus(string.format("Loop %d | Panen ke-%d (titik %d)", loopCount + 1, stats.harvested, i))
                    waitInterruptible(config.eDelay, myId)
                end
            end

            if not alive(myId) then break end

            loopCount += 1
            stats.loops += 1
            updateStats()
            if config.maxLoops > 0 and loopCount >= config.maxLoops then
                stopRun(string.format("Selesai (%d loop)", loopCount))
                return
            end

            highlight(nil)
            setStatus(string.format("Loop %d selesai, jeda %ss", loopCount, config.loopDelay))
            waitInterruptible(config.loopDelay, myId)
        end
    end)
end

btnStart.MouseButton1Click:Connect(startRun)
