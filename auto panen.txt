--[[
    🌾 AUTO PANEN + AUTO SIMPAN KE PETI (Trunk) - versi drag & drop
    - Titik Panen : player berjalan (WASD) ke titik, tunggu 1 detik, lalu panen
                    (mode 2 = klik lewat TOMBOL BULAT melayang yang digeser ke tombol
                    panen di layar, lalu dikunci; mode 1 = tekan tombol F), lanjut
    - Titik Jalan : player hanya melewati titik
    - Auto simpan berjalan berkala: tiap N menit (pengaturan "Simpan otomatis tiap", tab Peti)
    - Penghitung panen membaca teks notifikasi di layar ("Harvested ... xN"), bukan jumlah tombol ditekan
    - Urutan rute BOLAK-BALIK: 1-2-3-4-5-6 lalu 5-4-3-2-1, lalu ulang (tanpa melompat ke titik 1)
    - Saat PANEN: karakter hanya bergerak di GARIS RUTE antar titik (tidak melenceng).
      Jika tersangkut, mundur ke titik sebelumnya lalu coba lagi di jalur yang sama.
    - Kalibrasi Drag (tab Peti) sekarang 4 langkah:
        1) klik slot item pertama, 2) klik area peti (tujuan drop),
        3) (opsional) klik slot item ke-2, 4) klik POSISI tombol Confirm.
      Setelah tiap seretan item, skrip mengklik posisi Confirm yang sudah dikalibrasi
      (tidak mencari lewat teks lagi). Di langkah 4 skrip menyeret 1 item otomatis agar
      pop up Confirm muncul; jika belum muncul tekan "Seret lagi" di banner.
    - Waktu simpan tiba -> HITUNG RUTE dulu (Pathfinding, 8 titik berdiri di sekeliling peti,
      pilih yang valid & terpendek), baru jalan ke peti. Jika tersangkut di tengah jalan, karakter
      kembali ke jalur panen dan mengikuti titik-titik rute sampai lepas,
      lalu mencoba mendekati peti lagi dari titik rute tersebut.
    - Scan peti: nama default "trunk, chest" (otomatis kena "Mimic Chest").
      Tombol "Cek Objek Sekitar" membantu mencari nama chest yang belum diketahui.
    - Perlu KALIBRASI DRAG sekali (tab Peti)
]]

local function new(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props) do o[k] = v end
    o.Parent = parent
    return o
end

local function corner(parent, r)
    return new("UICorner", { CornerRadius = UDim.new(0, r or 8) }, parent)
end

local function makeButton(parent, text, color, pos, size, anchor)
    local b = new("TextButton", {
        Text = text,
        Font = Enum.Font.GothamBold,
        TextSize = 12,
        TextColor3 = COLORS.text,
        BackgroundColor3 = color,
        AutoButtonColor = true,
        BorderSizePixel = 0,
        Position = pos,
        Size = size,
        AnchorPoint = anchor or Vector2.new(0, 0),
    }, parent)
    corner(b, 6)
    return b
end

local function makeLabel(parent, text, pos, size, align, color, textSize)
    return new("TextLabel", {
        Text = text,
        Font = Enum.Font.Gotham,
        TextSize = textSize or 12,
        TextColor3 = color or COLORS.text,
        TextXAlignment = align or Enum.TextXAlignment.Left,
        BackgroundTransparency = 1,
        Position = pos,
        Size = size,
    }, parent)
end

local RPAD = PAD + 6

local function addSetting(parent, labelText, key, y, kind, boxW, placeholder)
    boxW = boxW or 64
    local lbl = makeLabel(parent, labelText, UDim2.new(0, PAD, 0, y),
        UDim2.new(1, -PAD - RPAD - boxW - 4, 0, 24), Enum.TextXAlignment.Left, COLORS.text, 11)
    lbl.TextTruncate = Enum.TextTruncate.AtEnd
    local box = new("TextBox", {
        Text = tostring(config[key]),
        PlaceholderText = placeholder or "",
        PlaceholderColor3 = COLORS.sub,
        Font = Enum.Font.GothamBold,
        TextSize = 11,
        TextColor3 = COLORS.text,
        BackgroundColor3 = COLORS.panel,
        BorderSizePixel = 0,
        ClearTextOnFocus = false,
        ClipsDescendants = true,
        Position = UDim2.new(1, -RPAD, 0, y),
        AnchorPoint = Vector2.new(1, 0),
        Size = UDim2.new(0, boxW, 0, 24),
    }, parent)
    corner(box, 6)
    box.FocusLost:Connect(function()
        if kind == "text" then
            config[key] = box.Text
        else
            local n = tonumber(box.Text)
            if n and n >= 0 then
                if kind == "int" then n = math.floor(n) end
                config[key] = n
            end
        end
        box.Text = tostring(config[key])
    end)
end

local function currentElapsed()
    local e = stats.elapsed
    if running and stats.runStart then e += os.clock() - stats.runStart end
    return e
end

local function updateStats()
    local e = math.floor(currentElapsed())
    local timeStr
    if e >= 3600 then
        timeStr = string.format("%d:%02d:%02d", e // 3600, (e % 3600) // 60, e % 60)

        timeStr = string.format("%02d:%02d", e // 60, e % 60)
    end
    statsLabel.Text = string.format("🌱 Panen: %d  |  🔁 Loop: %d  |  ⏱ %s",

end

local function parseKeywords(str)
    local out = {}
    for part in string.gmatch(str or "", "[^,]+") do
        local k = part:match("^%s*(.-)%s*$"):lower()
        if k ~= "" then table.insert(out, k) end
    end
    return out
end

local function nameMatches(name, kws)
    local n = name:lower()
    for _, k in ipairs(kws) do
        if n:find(k, 1, true) then return true end
    end
    return false
end

local function chestNameMatches(name, kws)
    if config.exactMatch == 1 then
        local n = name:lower()
        for _, k in ipairs(kws) do
            if n == k then return true end
        end
        return false
    end
    return nameMatches(name, kws)
end

local function isInsideCharacter(inst)
    for _, pl in ipairs(Players:GetPlayers()) do
        local ch = pl.Character
        if ch and (inst == ch or inst:IsDescendantOf(ch)) then return true end
    end
    return false
end

local function getPart(inst)
    if inst:IsA("BasePart") then return inst end
    if inst:IsA("Model") then
        return inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true)
    end
end

local function boundsOf(inst)
    if inst:IsA("Model") then
        local ok, cf, size = pcall(function() return inst:GetBoundingBox() end)
        if ok and cf and size then return cf, size end
    elseif inst:IsA("BasePart") then
        return inst.CFrame, inst.Size
    end
end

local function distToBox(cf, size, pos)
    local lp = cf:PointToObjectSpace(pos)
    local h = size / 2
    local c = Vector3.new(
        math.clamp(lp.X, -h.X, h.X),
        math.clamp(lp.Y, -h.Y, h.Y),
        math.clamp(lp.Z, -h.Z, h.Z))
    return (lp - c).Magnitude
end

local function chestPos(chest)
    local cf = boundsOf(chest.inst)
    if cf then return cf.Position end
    return chest.part.Position
end

local function chestStandDist(chest)
    local _, size = boundsOf(chest.inst)
    local d = math.max(2, config.chestDist)
    if size then
        d = math.max(d, math.max(size.X, size.Z) / 2 + 2)
    end
    return d
end

local function chestValid(chest)

local function scanChests()
    local info = { outside = 0, nearestOutside = nil, unloaded = 0 }
    local _, _, root = getChar()
    if not root then setStatus("Karakter tidak ditemukan"); return 0, info end

    local kws = parseKeywords(config.chestKeywords)
    if #kws == 0 then setStatus("Isi nama peti dulu"); return 0, info end

    for i = #chests, 1, -1 do
        if not chestValid(chests[i]) then
            removeMarker(chests[i])
            table.remove(chests, i)
        end
    end
    local known = {}
    for _, c in ipairs(chests) do known[c.inst] = true end

    -- 1) kumpulkan semua objek yang namanya cocok
    local all = {}
    for _, inst in ipairs(workspace:GetDescendants()) do
        if (inst:IsA("Model") or inst:IsA("BasePart"))
            and chestNameMatches(inst.Name, kws) and not isInsideCharacter(inst) then
            table.insert(all, inst)
        end
    end

    -- 2) Model yang di dalamnya ada Model cocok lain dianggap grup, dilewati
    local keep, keepSet = {}, {}
    for _, m in ipairs(all) do
        local isGroup = false
        if m:IsA("Model") then
            for _, o in ipairs(all) do
                if o ~= m and o:IsA("Model") and o:IsDescendantOf(m) then
                    isGroup = true
                    break
                end
            end
        end
        if not isGroup then
            table.insert(keep, m)
            keepSet[m] = true
        end
    end

    -- 3) buang yang berada di dalam objek lain yang sudah dipilih
    local final = {}
    for _, m in ipairs(keep) do
        local nested = false
        local p = m.Parent
        while p and p ~= workspace do
            if keepSet[p] then nested = true break end
            p = p.Parent
        end
        if not nested then table.insert(final, m) end
    end

    -- 4) saring berdasarkan jarak ke kotak batas
    local found = {}
    print(string.format("[AutoPanen] Scan: %d kandidat nama cocok, radius %s stud", #final, tostring(config.chestScanRadius)))
    for _, inst in ipairs(final) do
        if not known[inst] then
            local part = getPart(inst)
            if not part then
                info.unloaded += 1
                print("[AutoPanen]  - " .. inst:GetFullName() .. " | belum ter-load (tidak ada part)")
            else
                local cf, size = boundsOf(inst)
                local d
                if cf then
                    d = distToBox(cf, size, root.Position)
                else
                    d = (part.Position - root.Position).Magnitude
                end
                print(string.format("[AutoPanen]  - %s | jarak %.1f stud", inst:GetFullName(), d))
                if d <= config.chestScanRadius then
                    table.insert(found, { inst = inst, part = part, dist = d })
                else
                    info.outside += 1
                    if not info.nearestOutside or d < info.nearestOutside then
                        info.nearestOutside = d
                    end
                end
            end
        end
    end

    table.sort(found, function(a, b) return a.dist < b.dist end)

    for _, f in ipairs(found) do
        local chest = { inst = f.inst, part = f.part, number = nextChestNo, full = false }
        nextChestNo += 1
        makeMarker(chest)
        table.insert(chests, chest)
    end

    refreshChestList()
    updateChestInfo()
    return #found, info
end

local function resetChests()
    for _, c in ipairs(chests) do removeMarker(c) end
    chests = {}
    nextChestNo = 1
    depositCooldown = 0
    refreshChestList()
    updateChestInfo()
end

    local extra = ""

    end
    if info.unloaded > 0 then
        extra = extra .. string.format(" | belum ter-load: %d", info.unloaded)
    end
    if n > 0 then
        setStatus(string.format("%d peti baru (total %d)%s", n, #chests, extra))
    elseif #chests == 0 then
        setStatus("Tidak ada peti ditemukan" .. extra)
    else
        setStatus("Tidak ada peti baru" .. extra)
    end
end)

-- Bantu cari nama chest yang belum diketahui:

local function inspectNearby()
    local _, _, root = getChar()
    if not root then setStatus("Karakter tidak ditemukan"); return end
    local seen, names = {}, {}
    print("[AutoPanen] === Objek interaktif dalam 40 stud ===")
    for _, d in ipairs(workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") or d:IsA("ClickDetector") then
            local part = d.Parent
            if part and part:IsA("Attachment") then part = part.Parent end
            if part and part:IsA("BasePart") and (part.Position - root.Position).Magnitude <= 40 then
                local owner = part:FindFirstAncestorOfClass("Model") or part
                if not seen[owner] then
                    seen[owner] = true
                    table.insert(names, owner.Name)
                    local extra = ""
                    if d:IsA("ProximityPrompt") then
                        extra = string.format(" | ObjectText='%s' ActionText='%s'", d.ObjectText, d.ActionText)
                    end
                    print(string.format("[AutoPanen]  - %s | %.0f stud%s",
                        owner:GetFullName(), (part.Position - root.Position).Magnitude, extra))
                end
            end
        end
    end
    if #names == 0 then
        setStatus("Tidak ada objek interaktif dalam 40 stud")
    else
        setStatus("Objek: " .. table.concat(names, ", ") .. " (lihat F9)")
    end
end

local function insideMain(pos)
    local function inside(f)
        if not f.Visible then return false end
        local p, s = f.AbsolutePosition, f.AbsoluteSize
        return pos.X >= p.X and pos.X <= p.X + s.X and pos.Y >= p.Y and pos.Y <= p.Y + s.Y
    end
    return inside(main) or inside(calibBanner) -- klik di panel/banner script diabaikan
end

        return Vector2.new(input.Position.X, input.Position.Y + GuiService:GetGuiInset().Y)

end

local AMOUNT_NAMES = { "Amount", "Count", "Quantity", "Stack" }

local function countTool(tool)
    for _, n in ipairs(AMOUNT_NAMES) do
        local v = tool:GetAttribute(n)
        if type(v) == "number" then return v end
    end
    for _, n in ipairs(AMOUNT_NAMES) do
        local v = tool:FindFirstChild(n)
        if v and (v:IsA("IntValue") or v:IsA("NumberValue")) then return v.Value end
    end
    return 1
end

local function allTools()
    local list = {}
    local bp = player:FindFirstChildOfClass("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") then table.insert(list, t) end
        end
    end
    local char = player.Character
    if char then
        for _, t in ipairs(char:GetChildren()) do
            if t:IsA("Tool") then table.insert(list, t) end
        end
    end
    return list
end

local function collectMaterials(skipTool)
    local matKw  = parseKeywords(config.materialKeywords)
    local keepKw = parseKeywords(config.keepKeywords)
    local out = {}
    for _, t in ipairs(allTools()) do
        local ok = true
        if #matKw > 0 then
            ok = nameMatches(t.Name, matKw)
        elseif skipTool and t == skipTool then
            ok = false
        end
        if ok and #keepKw > 0 and nameMatches(t.Name, keepKw) then ok = false end
        if ok then table.insert(out, t) end
    end
    return out
end

local function countMaterials(skipTool)
    if CUSTOM_MATERIAL_COUNT then
        local ok, n = pcall(CUSTOM_MATERIAL_COUNT)
        return ok and tonumber(n) or 0
    end
    local total = 0
    for _, t in ipairs(collectMaterials(skipTool)) do total += countTool(t) end
    return total
end

local function steerTo(root, target)
    local cam = workspace.CurrentCamera
    local dir = Vector3.new(target.X - root.Position.X, 0, target.Z - root.Position.Z)
    if dir.Magnitude < 0.05 or not cam then
        releaseMoveKeys()
        return
    end
    dir = dir.Unit

    local look = cam.CFrame.LookVector
    look = Vector3.new(look.X, 0, look.Z)
    local right = cam.CFrame.RightVector
    right = Vector3.new(right.X, 0, right.Z)
    if look.Magnitude < 0.05 then look = Vector3.new(0, 0, -1) else look = look.Unit end
    if right.Magnitude < 0.05 then right = Vector3.new(1, 0, 0) else right = right.Unit end

    local f = dir:Dot(look)
    local r = dir:Dot(right)

    setKey(KC.W, f > 0.3)
    setKey(KC.S, f < -0.3)
    setKey(KC.D, r > 0.3)
    setKey(KC.A, r < -0.3)
end

        task.wait(0.08)
        VIM:SendMouseButtonEvent(from.X, from.Y, 0, true, game, 0)
        task.wait(0.12)

        for i = 1, steps do
            local p = from:Lerp(to, i / steps)
            VIM:SendMouseMoveEvent(p.X, p.Y, game)
            task.wait(0.02)
        end
        task.wait(0.08)
        VIM:SendMouseButtonEvent(to.X, to.Y, 0, false, game, 0)
    end)
end

--[[
    driveTo(target, myId, opts)
    opts.radius     : jarak dianggap sampai
    opts.timeout    : batas waktu (detik)
    opts.stuckLimit : jika diisi, return "stuck" setelah sekian detik tanpa gerak
                      (jika kosong: hanya lompat & terus mencoba sampai timeout)
    opts.line       : {a, b} -> karakter dipaksa mengikuti GARIS a->b
                      (membidik titik di depan pada garis, jadi tidak melenceng)
    Return: "reached" | "stuck" | "timeout" | "stopped"
]]
local function driveTo(target, myId, opts)
    opts = opts or {}
    local radius     = opts.radius or config.radius
    local timeout    = opts.timeout or config.stepTimeout
    local stuckLimit = opts.stuckLimit
    local line       = opts.line

    local start = os.clock()
    local lastPos, lastCheck = nil, os.clock()
    local stuckFor, jumpRelease = 0, nil
    local result = "timeout"

    while alive(myId) do
        local _, hum, root = getChar()
        if not root then result = "stopped" break end
        local pos = root.Position

        if flatDist(pos, target) <= radius and math.abs(pos.Y - target.Y) < 12 then
            result = "reached"
            break
        end
        if os.clock() - start > timeout then
            result = "timeout"
            break
        end

        -- titik bidik: di garis rute (kalau ada), selain itu langsung ke target
        local aim = target
        if line then
            local a, b = line[1], line[2]
            local ab = Vector3.new(b.X - a.X, 0, b.Z - a.Z)
            local len = ab.Magnitude
            if len > 0.1 then
                local dir = ab / len
                local t = math.clamp(Vector3.new(pos.X - a.X, 0, pos.Z - a.Z):Dot(dir), 0, len)
                local ahead = math.min(len, t + config.lookahead)
                aim = Vector3.new(a.X + dir.X * ahead, b.Y, a.Z + dir.Z * ahead)
            end
        end

        if config.useWASD == 1 then
            steerTo(root, aim)
        else
            hum:MoveTo(aim)
        end

        -- deteksi nyangkut tiap 1 detik
        if os.clock() - lastCheck >= 1 then
            if lastPos and (pos - lastPos).Magnitude < 0.6 then
                stuckFor += 1
                if config.useWASD == 1 then
                    setKey(KC.Space, true)
                    jumpRelease = os.clock() + 0.2
                else
                    hum.Jump = true
                end
            else
                stuckFor = 0
            end
            lastPos = pos
            lastCheck = os.clock()
        end
        if jumpRelease and os.clock() >= jumpRelease then
            setKey(KC.Space, false)
            jumpRelease = nil
        end
        if stuckLimit and stuckFor >= stuckLimit then
            result = "stuck"
            break
        end
        task.wait(0.05)
    end

    if not (opts.keepKeys and result == "reached") then releaseMoveKeys() end
    return result
end

-- Berjalan dari titik a ke b HANYA di garis rute. Jika gagal: mundur ke a,
-- lalu coba lagi di jalur yang sama. Return "reached" | "failed" | "stopped"
local function walkRoute(a, b, myId)
    local tries = math.max(1, config.retryPerPoint)
    for attempt = 1, tries do
        local r = driveTo(b, myId, {
            radius = config.radius, timeout = config.stepTimeout, line = { a, b },
        })
        if r == "reached" then return "reached" end
        if r == "stopped" or not alive(myId) then return "stopped" end

        if attempt < tries then
            setStatus(string.format("Tersangkut, mundur di jalur & coba lagi (%d/%d)", attempt, tries))
            local rr = driveTo(a, myId, {
                radius = config.radius + 1, timeout = config.stepTimeout, line = { b, a },
            })
            if rr == "stopped" or not alive(myId) then return "stopped" end
        end
    end
    return "failed"
end

local function nearestIdx(pos)
    local best, bd
    for i, p in ipairs(points) do
        local d = flatDist(p.pos, pos)
        if not bd or d < bd then best, bd = i, d end
    end
    return best
end

--[[
    travel: menuju target (peti / kembali ke rute) secara langsung.
    Jika tersangkut -> kembali ke jalur panen, ikuti titik-titik rute menuju
    titik rute terdekat dengan target, dan setelah tiap titik coba lagi
    mendekati target dari situ.
    fromIdx: titik rute awal untuk fallback (default: terdekat dengan karakter)
    Return: "reached" | "failed" | "stopped"
]]
local function travel(target, myId, radius, fromIdx)
    radius = radius or config.radius
    local function direct()
        return driveTo(target, myId, {
            radius = radius, timeout = config.stepTimeout, stuckLimit = config.stuckLimit,
        })
    end

    local r = direct()
    if r == "reached" or r == "stopped" then return r end

    local n = #points
    if n == 0 then return "failed" end
    local _, _, root = getChar()
    if not root then return "stopped" end

    setStatus("Tersangkut, ikuti jalur panen...")
    local cur = fromIdx or nearestIdx(root.Position)
    local to  = nearestIdx(target)

    -- kembali dulu ke jalur panen
    local back = driveTo(points[cur].pos, myId, {
        radius = config.radius + 1, timeout = config.stepTimeout,
    })
    if back == "stopped" or not alive(myId) then return "stopped" end

    -- rute bolak-balik (bukan melingkar): ikuti titik secara berurutan tanpa lompat dari N ke 1
    local step = (to >= cur) and 1 or -1
    local count = math.abs(to - cur)

    for hop = 0, count do
        if hop > 0 then
            local nxt = cur + step
            local w = walkRoute(points[cur].pos, points[nxt].pos, myId)
            if w == "stopped" then return "stopped" end
            cur = nxt
        end

        setStatus(string.format("Dari titik rute %d, coba mendekati tujuan", cur))
        local d = direct()
        if d == "reached" or d == "stopped" then return d end

        -- gagal lagi: balik ke jalur sebelum lanjut
        local bk = driveTo(points[cur].pos, myId, {
            radius = config.radius + 1, timeout = config.stepTimeout,
        })
        if bk == "stopped" or not alive(myId) then return "stopped" end
    end
    return "failed"
end

------------------------------------------------------------------
-- HITUNG RUTE KE PETI (Pathfinding) - dihitung & divalidasi dulu, baru berangkat
------------------------------------------------------------------
-- titik berdiri di sekeliling peti (8 arah), diurutkan dari yang terdekat dengan karakter
local function chestGoals(chest)
    local cf, size = boundsOf(chest.inst)
    local c = cf and cf.Position or chest.part.Position
    local bottom = size and (c.Y - size.Y / 2) or (c.Y - 2)
    local d = chestStandDist(chest) + 0.5
    local goals = {}
    for k = 0, 7 do
        local a = math.rad(k * 45)
        table.insert(goals, Vector3.new(c.X + math.cos(a) * d, bottom + 2.5, c.Z + math.sin(a) * d))
    end
    local _, _, root = getChar()
    if root then
        table.sort(goals, function(a, b)
            return flatDist(a, root.Position) < flatDist(b, root.Position)
        end)
    end
    return goals
end

-- hitung rute ke beberapa tujuan, pilih yang valid dan terpendek
-- return: waypoint, panjang (atau nil jika tidak ada rute yang valid)
local function bestPath(goals, myId)
    local _, _, root = getChar()
    if not root then return nil end
    local best, bestLen
    local found = 0
    for _, g in ipairs(goals) do
        if not alive(myId) then return nil end
        local path = PathfindingService:CreatePath({
            AgentRadius = 2, AgentHeight = 5, AgentCanJump = true, WaypointSpacing = 5,
        })
        local ok = pcall(function() path:ComputeAsync(root.Position, g) end)
        if ok and path.Status == Enum.PathStatus.Success then
            local wps = path:GetWaypoints()
            if #wps >= 1 and flatDist(wps[#wps].Position, g) <= 6 then
                local len = 0
                for i = 2, #wps do len += (wps[i].Position - wps[i - 1].Position).Magnitude end
                if not bestLen or len < bestLen then best, bestLen = wps, len end
                found += 1
                if found >= 4 then break end
            end
        end
    end
    return best, bestLen
end

-- ikuti waypoint satu per satu. return "done" | "blocked" | "stopped"
local function followWaypoints(wps, myId)
    for k = 1, #wps do
        if not alive(myId) then return "stopped" end
        local wp = wps[k]
        if wp.Action == Enum.PathWaypointAction.Jump then
            setKey(KC.Space, true)
            task.delay(0.25, function() setKey(KC.Space, false) end)
        end
        local r = driveTo(wp.Position, myId, {
            radius = 2.5, timeout = 6, stuckLimit = 3, keepKeys = (k < #wps),
        })
        if r == "stopped" then return "stopped" end
        if r ~= "reached" then return "blocked" end
    end
    return "done"
end

-- hitung rute dulu; jika valid baru berjalan. Jika terhalang, hitung ulang dari posisi sekarang
-- (maks 3 kali). Jika tetap gagal, pakai cara lama (ikuti jalur panen).
local function smartTravel(goals, finalTarget, myId, radius, fromIdx)
    if config.usePathfind == 1 then
        for attempt = 1, 3 do
            if not alive(myId) then return "stopped" end
            setStatus(string.format("Menghitung rute yang benar (%d/3)...", attempt))
            local wps, len = bestPath(goals, myId)
            if not wps then break end
            setStatus(string.format("Rute valid (%d titik, %.0f stud), berangkat", #wps, len or 0))

            local r = followWaypoints(wps, myId)
            if r == "stopped" then return "stopped" end
            local _, _, root = getChar()
            if not root then return "stopped" end

            if flatDist(root.Position, finalTarget) <= radius + 3 then return "reached" end
            if r == "done" then
                -- sudah di ujung rute, dekati tujuan sedikit lagi
                local d = driveTo(finalTarget, myId, {
                    radius = radius, timeout = 5, stuckLimit = config.stuckLimit,
                })
                if d == "reached" or d == "stopped" then return d end
            end
            setStatus("Terhalang, hitung ulang rute...")
        end
        setStatus("Rute tidak ditemukan, pakai jalur panen")
    end
    return travel(finalTarget, myId, radius, fromIdx)
end

local function pickChest(skip)
    for _, c in ipairs(chests) do
        if not c.full and chestValid(c) and not skip[c] then return c end
    end
end

local function slotPos(idx)
    if calib.step and idx > 1 then
        return calib.slot + calib.step * (idx - 1)
    end
    return calib.slot
end

-- Return: "ok", "allfull", "unreachable", "nochest", "nocalib", "stopped"
local function depositAll(myId)
    local char, hum, root = getChar()
    if not root then return "stopped" end

    if not (calib.slot and calib.drop) then
        setStatus("Atur Kalibrasi Drag dulu (tab Peti)")
        depositCooldown = os.clock() + 30
        return "nocalib"
    end

    setStatus("Mencari peti terdekat...")
    scanChests()
    if #chests == 0 then return "nochest" end

    local returnPos = root.Position
    local leftIdx = routeIdx
    local skipTool = char:FindFirstChildOfClass("Tool")
    local result = "ok"
    local deposited = 0
    local skip = {}
    -- jika inventory game bukan Tool, isi tas tidak bisa dihitung -> mode seret buta
    local countable = countMaterials(skipTool) > 0

    while alive(myId) do
        if countable and countMaterials(skipTool) <= 0 then break end

        local chest = pickChest(skip)
        if not chest then
            result = next(skip) and "unreachable" or "allfull"
            break
        end

        setStatus(string.format("Menuju peti #%d", chest.number))
        local tr = smartTravel(chestGoals(chest), chestPos(chest), myId, chestStandDist(chest), leftIdx)
        if not alive(myId) then break end
        if tr ~= "reached" then
            skip[chest] = true
            setStatus(string.format("Peti #%d tidak terjangkau, coba peti lain", chest.number))
            continue
        end

        setStatus(string.format("Membuka peti #%d (%s)", chest.number, config.openKey))
        pressKey(openKeyCode(), 0.1)
        waitInterruptible(config.openDelay, myId)

        if not countable then
            -- mode buta: seret item sebanyak dragTotal kali, berhenti jika notifikasi peti penuh muncul
            local total = math.max(1, config.dragTotal)
            chestFullFlag = false
            for k = 1, total do
                if not alive(myId) then break end
                local sidx = (calib.step and config.slotCount > 1) and ((k - 1) % config.slotCount + 1) or 1
                setStatus(string.format("Pindah item ke peti #%d (%d/%d)", chest.number, k, total))
                dragMouse(slotPos(sidx), calib.drop)
                clickConfirm(myId)
                waitInterruptible(config.depositDelay, myId)
                deposited += 1
                if chestFullFlag then
                    chest.full = true
                    updateMarker(chest)
                    updateChestInfo()
                    refreshChestList()
                    setStatus(string.format("Peti #%d penuh, pindah peti berikutnya", chest.number))
                    break
                end
            end
            if config.closeChest == 1 and alive(myId) then
                pressKey(openKeyCode(), 0.1)
                task.wait(0.3)
            end
            if chest.full then continue end -- lanjut peti berikutnya
            break
        end

        local idx, fails = 1, 0
        while alive(myId) and not chest.full do
            local before = countMaterials(skipTool)
            if before <= 0 then break end

            setStatus(string.format("Pindah item ke peti #%d (slot %d)", chest.number, idx))
            dragMouse(slotPos(idx), calib.drop)
            clickConfirm(myId)
            waitInterruptible(config.depositDelay, myId)
            local after = countMaterials(skipTool)

            if after < before then
                local n = before - after
                deposited += n
                stats.deposited += n
                fails = 0
                updateChestInfo()
            else
                fails += 1
                if fails >= 2 then
                    if calib.step and idx < config.slotCount then
                        idx += 1
                        fails = 0
                    else
                        chest.full = true
                        updateMarker(chest)
                        updateChestInfo()
                        refreshChestList()
                        setStatus(string.format("Peti #%d penuh, pindah peti berikutnya", chest.number))
                    end
                end
            end
        end

        if config.closeChest == 1 and alive(myId) then
            pressKey(openKeyCode(), 0.1)
            task.wait(0.3)
        end
    end

    -- kembali ke jalur panen (titik rute terakhir), mengikuti rute bila tersangkut
    if alive(myId) then
        setStatus("Kembali ke jalur panen...")
        local dest = (leftIdx and points[leftIdx]) and points[leftIdx].pos or returnPos
        smartTravel({ dest }, dest, myId, config.radius + 1, nil)
    end

    if not alive(myId) then return "stopped" end
    if deposited == 0 and result ~= "allfull" then
        depositCooldown = os.clock() + 20
    end
    return result
end

local function parseHarvestAmount(text)
    if config.countAmount ~= 1 then return 1 end
    local n = text:match("%f[%a][xX]%s*(%d[%d,]*)")      -- "x3"
        or text:match("(%d[%d,]*)%s*[xX]%f[%A]")         -- "3x"
        or text:match("×%s*(%d[%d,]*)")                  -- "×3"
        or text:match("(%d[%d,]*)%s*×")                  -- "3×"
    if n then n = tonumber((n:gsub(",", ""))) end
    return n or 1
end

local kwCache = {}
local function kwOf(key)
    local s = tostring(config[key] or "")
    local c = kwCache[key]
    if not c or c.s ~= s then
        c = { s = s, list = parseKeywords(s) }
        kwCache[key] = c
    end
    return c.list
end

-- label benar-benar tampil di layar (bukan template tersembunyi)
local function isShown(l)
    if not l.Visible or l.TextTransparency >= 0.9 then return false end
    local p = l.Parent
    while p do
        if p:IsA("GuiObject") and not p.Visible then return false end
        if p:IsA("ScreenGui") then return p.Enabled end
        p = p.Parent
    end
    return true
end

-- deteksi notifikasi peti penuh
local function scanFlags(label, text, allowChest)
    if not isShown(label) then return end
    if allowChest and nameMatches(text, kwOf("chestFullNotif")) then
        chestFullFlag = true
    end
end

local function checkNotif(label)
    if not running or not gui.Parent then return end
    if not label.Visible or label:IsDescendantOf(gui) then return end
    local raw = label.Text
    if raw == "" then return end
    if #raw > 120 then return end
    local text = raw:gsub("<[^>]+>", "")                 -- buang tag rich text
    scanFlags(label, text, true)
    local kw = tostring(config.harvestNotif or ""):lower()
    if kw == "" or not text:lower():find(kw, 1, true) then return end

    if seenText[label] == raw then return end            -- teks yang sama di label yang sama
    seenText[label] = raw
    local now = os.clock()
    if lastCounted[text] and now - lastCounted[text] < 0.25 then return end -- label ganda/bayangan
    lastCounted[text] = now

    stats.harvested += parseHarvestAmount(text)
    updateStats()
end

local function watchLabel(l)
    if not (l:IsA("TextLabel") or l:IsA("TextButton")) then return end
    l:GetPropertyChangedSignal("Text"):Connect(function() checkNotif(l) end)
    checkNotif(l) -- label baru yang sudah berisi teks
end

task.spawn(function()
    local pg = player:FindFirstChildOfClass("PlayerGui") or player:WaitForChild("PlayerGui")
    notifAdded = pg.DescendantAdded:Connect(watchLabel)
    for _, d in ipairs(pg:GetDescendants()) do watchLabel(d) end
end)

local function updateButtons()
    btnStart.BackgroundColor3 = running and COLORS.grey or COLORS.green
    btnStop.BackgroundColor3  = running and COLORS.red or COLORS.grey
end
updateButtons()

local function stopRun(msg)
    if running and stats.runStart then
        stats.elapsed += os.clock() - stats.runStart
    end
    stats.runStart = nil
    running = false
    runId += 1
    releaseMoveKeys()
    local _, hum, root = getChar()
    if hum and root then hum:MoveTo(root.Position) end
    highlight(nil)
    updateButtons()
    setStatus(msg or "Berhenti")
end

local function startRun()
    if running then return end
    if calibStep > 0 then setStatus("Selesaikan kalibrasi dulu"); return end
    if #points == 0 then setStatus("Tambahkan titik dulu!"); return end

    running = true
    runId += 1
    local myId = runId
    routeIdx = nil
    storeNow = false
    nextStoreAt = os.clock() + math.max(0.1, config.storeInterval) * 60
    stats.runStart = os.clock()
    updateButtons()

    task.spawn(function()
        local loopCount = 0
        while alive(myId) do
            -- urutan bolak-balik: 1..N lalu N-1..1 (contoh 1-2-3-4-5-6-5-4-3-2-1).
            -- siklus berikutnya mulai dari titik 2 karena karakter sudah berada di titik 1.
            local order = {}
            local n = #points
            for k = (loopCount == 0) and 1 or math.min(2, n), n do table.insert(order, k) end
            for k = n - 1, 1, -1 do table.insert(order, k) end

            for _, i in ipairs(order) do
                local pt = points[i]
                if not alive(myId) then break end

                if storeNow or (config.autoStore and config.storeInterval > 0 and os.clock() >= nextStoreAt) then
                    local manual = storeNow
                    storeNow = false
                    setStatus(manual and "Simpan sekarang, menuju peti..." or "Waktunya simpan otomatis, menuju peti...")
                    local res = depositAll(myId)
                    nextStoreAt = os.clock() + config.storeInterval * 60
                    if res == "allfull" then
                        stopRun("Semua peti penuh, script berhenti")
                        return
                    elseif res == "nochest" then
                        setStatus("Waktunya simpan tapi tidak ada peti terdeteksi")
                    elseif res == "nocalib" then
                        setStatus("Waktunya simpan tapi Kalibrasi Drag belum diatur")
                    elseif res == "unreachable" then
                        setStatus("Peti tidak terjangkau, lanjut panen")
                    end
                    if not alive(myId) then break end
                end

                highlight(i)
                setStatus(string.format("Loop %d | Menuju titik %d/%d", loopCount + 1, i, #points))

                -- MODE PANEN: hanya lewat garis rute (titik sebelumnya -> titik ini)
                local r
                if routeIdx and points[routeIdx] then
                    r = walkRoute(points[routeIdx].pos, pt.pos, myId)
                else
                    r = travel(pt.pos, myId, config.radius) -- awal: belum ada titik acuan
                end
                if not alive(myId) then break end

                if r == "reached" then
                    routeIdx = i
                    if pt.type == "action" then
                        setStatus(string.format("Tiba di titik %d, tunggu %ss lalu panen", i, tostring(config.preHarvestDelay)))
                        waitInterruptible(config.preHarvestDelay, myId)
                        if not alive(myId) then break end
                        pressHarvest()
                        setStatus(string.format("Loop %d | Panen dijalankan (titik %d)", loopCount + 1, i))
                        waitInterruptible(config.eDelay, myId)
                    end
                else
                    setStatus(string.format("Titik %d gagal dicapai, lanjut titik berikutnya", i))
                end
            end

            if not alive(myId) then break end

            loopCount += 1
            stats.loops += 1
            updateStats()
            if config.maxLoops > 0 and loopCount >= config.maxLoops then
                stopRun(string.format("Selesai (%d loop)", loopCount))
                return
            end

            highlight(nil)
            setStatus(string.format("Loop %d selesai, jeda %ss", loopCount, config.loopDelay))
            waitInterruptible(config.loopDelay, myId)
        end
    end)
end

btnStart.MouseButton1Click:Connect(startRun)

    running = true
    runId += 1

    updateButtons()
    task.spawn(function()
        local res = depositAll(myId)

        local msgs = {

            allfull = "Semua peti penuh",
            nochest = "Tidak ada peti terdeteksi",
            nocalib = "Kalibrasi Drag belum diatur",
            unreachable = "Peti tidak terjangkau",
        }
        stopRun(msgs[res] or "Simpan selesai")
    end)
end)

    btnMin.Text = minimized and "+" or "–"

        TweenService:Create(main, TweenInfo.new(0.2, Enum.EasingStyle.Quad),
            { Size = UDim2.new(0, WIN_W, 0, TITLE_H + STRIP_H) }):Play()

            { Size = UDim2.new(0, WIN_W, 0, EXP_H) }):Play()
        task.delay(0.15, function()
            if not minimized then content.Visible = true end
        end)
    end
end
